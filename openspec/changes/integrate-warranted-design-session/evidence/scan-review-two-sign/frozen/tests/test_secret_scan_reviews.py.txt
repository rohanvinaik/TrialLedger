"""Reviewed public evidence never licenses another detector, file or changed value."""

import hashlib
import json
import subprocess

import pytest

from scripts import push_checks
from scripts.push_checks import secret_review_state


def test_exact_review_is_admitted():
    assert (
        secret_review_state("a.json", "example", "abc", [("a.json", "example", "abc")])
        == "reviewed_non_secret"
    )


@pytest.mark.parametrize(
    "path,detector,digest",
    [("b.json", "example", "abc"), ("a.json", "other", "abc"), ("a.json", "example", "changed")],
)
def test_review_must_not_extend_to_another_identity(path, detector, digest):
    assert secret_review_state(path, detector, digest, [("a.json", "example", "abc")]) == "unreviewed"


def test_empty_registry_does_not_admit_a_finding():
    assert secret_review_state("a.json", "example", "abc", []) == "unreviewed"


def test_reviewed_finding_still_runs_all_configured_detectors(monkeypatch):
    line = '  "reference": "public-example"'
    digest = hashlib.sha256(line.encode()).hexdigest()
    sha = "a" * 40
    commands = []
    monkeypatch.setattr(push_checks, "secret_reviews", lambda: [("a.json", "example", digest)])
    monkeypatch.setattr(push_checks, "commit_additions", lambda commit: {"a.json": [line]})

    def fake_run(command, cwd=None, env=None):
        commands.append(command)
        report = {
            "results": {
                sha + "/a.json": [
                    {"line_number": 1, "type": "example"},
                    {"line_number": 1, "type": "another-detector"},
                ]
            }
        }
        return subprocess.CompletedProcess(command, 0, json.dumps(report), "")

    monkeypatch.setattr(push_checks, "run", fake_run)
    assert push_checks.scan([sha]) == [
        f"a possible secret in {sha[:7]} a.json, added line 1: another-detector"
    ]
    assert commands[0].count("--disable-plugin") == 1
    assert "HexHighEntropyString" in commands[0]


@pytest.mark.parametrize(
    "value", [None, [], {}, {"version": True, "reviews": []}, {"version": 1, "reviews": [{}]}]
)
def test_malformed_review_registry_refuses_scan(tmp_path, monkeypatch, value):
    folder = tmp_path / "scripts"
    folder.mkdir()
    (folder / "secret_scan_reviews.json").write_text(json.dumps(value))
    monkeypatch.setattr(push_checks, "ROOT", tmp_path)
    with pytest.raises(push_checks.Unchecked):
        push_checks.secret_reviews()


def test_missing_review_registry_is_not_a_clean_scan(tmp_path, monkeypatch):
    monkeypatch.setattr(push_checks, "ROOT", tmp_path)
    with pytest.raises(push_checks.Unchecked):
        push_checks.secret_reviews()
