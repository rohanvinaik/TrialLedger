COMMANDS=[]

def command(label,args,sent=None):
    completed=subprocess.run(args,input=sent,capture_output=True,timeout=60)
    (ROOT/(label+".stdout")).write_bytes(completed.stdout)
    (ROOT/(label+".stderr")).write_bytes(completed.stderr)
    COMMANDS.append({"label":label,"argv":list(map(str,args)),"returncode":completed.returncode,"stdout_sha256":hashlib.sha256(completed.stdout).hexdigest(),"stderr_sha256":hashlib.sha256(completed.stderr).hexdigest()})
    (ROOT/"entrypoint-commands.json").write_text(json.dumps(COMMANDS,indent=2)+"\n")
    return completed

def normalized(value):
    return json.loads(json.dumps(value,allow_nan=False))

VERIFIED=[]
def check(label,request,expected_code="complete"):
    before=copy.deepcopy(request)
    expected=run(request)
    assert expected["code"]==expected_code,(label,expected)
    assert request==before,label
    path=ROOT/(label+"-request.json")
    path.write_text(json.dumps(request,ensure_ascii=False,allow_nan=False)+"\n")
    (ROOT/(label+"-result.json")).write_text(json.dumps(expected,ensure_ascii=False,allow_nan=False)+"\n")
    cli=command(label+"-cli",[ROOT/"runtime/bin/trialledger","reason","--request",path])
    assert cli.returncode==(0 if expected_code=="complete" else 2),(label,cli.stderr)
    assert json.loads(cli.stdout)==normalized(expected),label
    messages=[{"jsonrpc":"2.0","id":1,"method":"initialize","params":{"protocolVersion":"2025-11-25","capabilities":{},"clientInfo":{"name":"runtime-wheel-check","version":"1"}}},{"jsonrpc":"2.0","id":2,"method":"tools/call","params":{"name":"reason","arguments":{"request":request}}}]
    mcp=command(label+"-mcp",[ROOT/"runtime/bin/trialledger","mcp","--root",ROOT/"reports"],("".join(json.dumps(m,allow_nan=False)+"\n" for m in messages)).encode())
    assert mcp.returncode==0,(label,mcp.stderr)
    replies=[json.loads(line) for line in mcp.stdout.splitlines()]
    assert len(replies)==2 and replies[0]["id"]==1 and replies[1]["id"]==2,(label,replies)
    result=json.loads(replies[1]["result"]["content"][0]["text"])
    assert result.pop("tool")=="reason"
    assert result==normalized(expected),label
    assert request==before,label
    VERIFIED.append({"case":label,"operation":request["operation"],"code":expected_code,"transports":["executable CLI","stdio MCP"],"result_sha256":hashlib.sha256((ROOT/(label+"-result.json")).read_bytes()).hexdigest()})
    return expected
