def _required(identity: str, body: a.Literal) -> dict:
    rule = a.Rule("invented-candidate", "1", warrants=("user:invented-candidate",))
    assertion = a.ScopedAssertion(
        identity,
        "user:invented-candidate",
        "required",
        body,
        rule=rule,
        status="known",
        resolution="resolved",
    )
    return json.loads(a.canonical(assertion))


def example() -> dict:
    """Invented: adult ECOG <=2, with or without a marker requirement, after a design_bounds analysis."""
    participant = a.Subject("participant")
    age = a.Predicate("measurement", "example_age")
    scale = a.Predicate("measurement", "example_scale")
    marker = a.Predicate("demographic", "example_marker")
    adult = a.Literal(participant, age, a.Value("quantity", "ge", number=18.0, unit="years"))
    at_most_2 = a.Literal(participant, scale, a.Value("quantity", "le", number=2.0, unit="example_scale"))
    has_marker = a.Literal(participant, marker, a.Value("boolean", "eq"))

    def plain(model: object) -> dict:
        return json.loads(a.canonical(model))

    return {
        "operation": "evidence_requests",
        "study_ref": "invented:evidence-request-example",
        "candidates": [
            {
                "id": "adult_scale_le_2",
                "assertions": [_required("adult", adult), _required("scale", at_most_2)],
            },
            {
                "id": "adult_scale_le_2_marker",
                "assertions": [
                    _required("adult-m", adult),
                    _required("scale-m", at_most_2),
                    _required("marker-m", has_marker),
                ],
            },
        ],
        "survivors": ["adult_scale_le_2", "adult_scale_le_2_marker"],
        "survival_basis": ["invented:design_bounds remaining"],
        "variables": [
            {
                "id": "scale",
                "subject": plain(participant),
                "predicate": plain(scale),
                "time": None,
                "domain": {"kind": "integer", "unit": "example_scale", "minimum": 0, "maximum": 4},
            },
            {
                "id": "marker",
                "subject": plain(participant),
                "predicate": plain(marker),
                "time": None,
                "domain": {"kind": "boolean"},
            },
            {
                "id": "age",
                "subject": plain(participant),
                "predicate": plain(age),
                "time": None,
                "domain": {
                    "kind": "continuous",
                    "unit": "years",
                    "minimum": 0,
                    "maximum": None,
                    "cut_points": [],
                },
            },
        ],
    }
